```tsx
import { DatePicker, type DatePickerValue } from '@winniekagendo/componentiq';

function ReportFilter() {
  const [range, setRange] = React.useState<DatePickerValue>({
    startDate: null,
    endDate: null,
  });

  return (
    <DatePicker
      value={range}
      onChange={setRange}
      variant='range'
      placeholder='Select report period'
    />
  );
}
```
