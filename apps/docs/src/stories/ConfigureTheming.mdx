import {
  ComponentIqProvider,
  type ComponentIqTokens,
} from '@winniekagendo/componentiq';

const brandTheme: ComponentIqTokens = {
  colors: {
    primary: '#8D493A',
    primaryForeground: '#FDFAF9',
    secondary: '#347887',
    secondaryForeground: '#FDFAF9',

    background: '#FDFAF9',
    surface: '#FFFFFF',
    secondaryBackground: '#F9FAFB',
    hover: '#F3F4F6',

    textPrimary: '#111827',
    foreground: '#374151',
    textSecondary: '#4B5563',
    textMuted: '#6B7280',
    textDisabled: '#9CA3AF',

    error: '#B42318',
    errorPastel: '#FEF3F2',
    warning: '#B54708',
    warningPastel: '#FFF4E5',
    success: '#067647',
    successPastel: '#ECFDF3',
    information: '#175CD3',
    informationPastel: '#EFF8FF',
    link: '#175CD3',
    linkPastel: '#EFF8FF',
  },
  radius: {
    sm: '0.25rem',
    md: '0.5rem',
    lg: '0.75rem',
  },
  spacing: {
    xs: '0.25rem',
    sm: '0.5rem',
    md: '1rem',
    lg: '1.5rem',
    xl: '2rem',
  },
  shadows: {
    sm: '0 1px 2px rgb(0 0 0 / 0.05)',
    md: '0 4px 12px rgb(0 0 0 / 0.08)',
    lg: '0 10px 30px rgb(0 0 0 / 0.12)',
  },
  typography: {
    fontFamily: "'Rubik', ui-sans-serif, system-ui, sans-serif",
    headingFontFamily: "'Rubik', ui-sans-serif, system-ui, sans-serif",
    baseSize: '1rem',
    headingWeight: 700,
    bodyWeight: 400,
  },
  strokeWidth: {
    hairline: '0.5px',
    thin: '1px',
    md: '1.5px',
    lg: '2px',
  },
};

export function AppTheme({ children }: { children: React.ReactNode }) {
  return (
    <ComponentIqProvider tokens={brandTheme}>{children}</ComponentIqProvider>
  );
}
